local function DisableArsenalAntiCheat()
	local count = 0
	for _, obj in ipairs(game:GetDescendants()) do
		if obj:IsA("BaseScript") then
			local n = obj.Name:lower()
			if n:find("anticheat") or n:find("anti_cheat") or n:find("anti-cheat")
				or n:find("anti cheat") or n == "ac" or n:find("sentry") then
				if obj:IsA("LocalScript") then
					pcall(function() obj.Disabled = true end)
					count = count + 1
				end
			end
		end
	end
	return count
end

local function hashString(str)
	local hash = 5381
	for i = 1, #str do
		hash = (bit32 and bit32.bxor(hash * 33, string.byte(str, i))) or ((hash * 33) + string.byte(str, i))
	end
	return string.format("%X", math.abs(hash))
end

local function createKey(durationUnit, durationVal)
	local randomPart = string.format("%04X%04X", math.random(0, 65535), math.random(0, 65535))
	local expireTime
	if durationUnit == "Lifetime" then
		expireTime = "LIFETIME"
	else
		local seconds = 0
		if durationUnit == "Jours" then seconds = durationVal * 86400
		elseif durationUnit == "Mois" then seconds = durationVal * 30 * 86400
		elseif durationUnit == "Années" then seconds = durationVal * 365 * 86400 end
		expireTime = tostring(os.time() + seconds)
	end
	local payload = "FK-" .. randomPart .. "-" .. expireTime
	local sig = hashString(payload .. SECRET_SALT)
	return payload .. "-" .. sig
end

local function verifyKey(keyStr)
	if not keyStr or type(keyStr) ~= "string" then return false, "Clé invalide" end
	keyStr = keyStr:gsub("%s+", "")
	if blockedKeys[keyStr] then return false, "Clé bloquée par l'Admin" end
	local parts = string.split(keyStr, "-")
	if #parts ~= 4 or parts[1] ~= "FK" then return false, "Format de clé incorrect" end
	local randomPart, expireStr, sig = parts[2], parts[3], parts[4]
	local expectedPayload = "FK-" .. randomPart .. "-" .. expireStr
	local expectedSig = hashString(expectedPayload .. SECRET_SALT)
	if sig ~= expectedSig then return false, "Clé invalide ou falsifiée" end
	if expireStr == "LIFETIME" then return true, "Lifetime (Infinie)", math.huge end
	local expireTime = tonumber(expireStr)
	if not expireTime then return false, "Clé corrompue" end
	local now = os.time()
	local remaining = expireTime - now
	if remaining <= 0 then return false, "Clé expirée", 0 end
	local days = math.floor(remaining / 86400)
	local hours = math.floor((remaining % 86400) / 3600)
	local mins = math.floor((remaining % 3600) / 60)
	local secs = remaining % 60
	local timeFormatted = ""
	if days > 0 then timeFormatted = timeFormatted .. days .. "j " end
	if hours > 0 or days > 0 then timeFormatted = timeFormatted .. hours .. "h " end
	if mins > 0 or hours > 0 or days > 0 then timeFormatted = timeFormatted .. mins .. "m " end
	timeFormatted = timeFormatted .. secs .. "s"
	return true, timeFormatted, remaining
end

local function copyToClipboard(text)
	if setclipboard then setclipboard(text)
	elseif toclipboard then toclipboard(text)
	elseif syn and syn.write_clipboard then syn.write_clipboard(text) end
end

local function CreateSlider(text, minVal, maxVal, default, callback, id)
	local SliderFrame = Instance.new("Frame")
	SliderFrame.Size = UDim2.new(1, 0, 0, 50); SliderFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
	SliderFrame.BorderSizePixel = 0; SliderFrame.LayoutOrder = nextOrder()
	Instance.new("UICorner", SliderFrame).CornerRadius = UDim.new(0, 6)

	local Label = Instance.new("TextLabel")

	Label.Font = Enum.Font.Gotham; Label.TextSize = 13
	Label.TextXAlignment = Enum.TextXAlignment.Left; Label.Parent = SliderFrame

	local ValueLabel = Instance.new("TextLabel")
	ValueLabel.Size = UDim2.new(0, 60, 0, 20); ValueLabel.Position = UDim2.new(1, -65, 0, 5)
	ValueLabel.BackgroundTransparency = 1; ValueLabel.Text = tostring(default)
	ValueLabel.TextColor3 = Color3.fromRGB(255, 60, 60); ValueLabel.Font = Enum.Font.GothamBold
	ValueLabel.TextSize = 13; ValueLabel.TextXAlignment = Enum.TextXAlignment.Right; ValueLabel.Parent = SliderFrame

	local SliderBar = Instance.new("TextButton")
	SliderBar.Size = UDim2.new(1, -20, 0, 6); SliderBar.Position = UDim2.new(0, 10, 0, 33)
	SliderBar.BackgroundColor3 = Color3.fromRGB(50, 50, 55); SliderBar.BorderSizePixel = 0
	SliderBar.Text = ""; SliderBar.AutoButtonColor = false; SliderBar.Parent = SliderFrame
	Instance.new("UICorner", SliderBar).CornerRadius = UDim.new(1, 0)

	local Fill = Instance.new("Frame")
	Fill.Size = UDim2.new((default - minVal) / (maxVal - minVal), 0, 1, 0)
	Fill.BackgroundColor3 = Color3.fromRGB(255, 30, 30); Fill.BorderSizePixel = 0; Fill.Parent = SliderBar
	Instance.new("UICorner", Fill).CornerRadius = UDim.new(1, 0)

	local sliderDragging = false
	local function setValue(v, fire)
		v = math.clamp(v, minVal, maxVal)
		local rel = (v - minVal) / (maxVal - minVal)
		Fill.Size = UDim2.new(rel, 0, 1, 0)
		ValueLabel.Text = tostring(math.floor(v + 0.5))
		if SliderRegistry[id] then SliderRegistry[id].value = v end
		if fire then callback(v) end
	end
	SliderRegistry[id] = {value = default, set = function(v) setValue(v, true) end}

		sliderDragging = true
		local rel = math.clamp((Mouse.X - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
		setValue(minVal + (maxVal - minVal) * rel, true)
	end)

		if sliderDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			local rel = math.clamp((input.Position.X - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
			setValue(minVal + (maxVal - minVal) * rel, true)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then sliderDragging = false end
	end)

	SliderFrame.Parent = ScrollingFrame
	return SliderFrame
end

local function setZoom(distance)
	if distance < LocalPlayer.CameraMinZoomDistance then
		LocalPlayer.CameraMinZoomDistance = distance; LocalPlayer.CameraMaxZoomDistance = distance
	else
		LocalPlayer.CameraMaxZoomDistance = distance; LocalPlayer.CameraMinZoomDistance = distance
	end
end
local function setOcclusion(mode) pcall(function() LocalPlayer.DevCameraOcclusionMode = mode end) end

local savedSettings = nil
local function restoreSettings()
	if not savedSettings then return end
	LocalPlayer.CameraMode = savedSettings.cameraMode
	LocalPlayer.CameraMaxZoomDistance = savedSettings.maxZoom
	LocalPlayer.CameraMinZoomDistance = math.clamp(savedSettings.minZoom, 0.5, savedSettings.maxZoom)
	setOcclusion(savedSettings.occlusion)
	Camera = Workspace.CurrentCamera
	if Camera then Camera.CameraType = savedSettings.cameraType end
	savedSettings = nil
end

local function onThirdPerson()
	if not thirdPersonActive then return end
	Camera = Workspace.CurrentCamera; if not Camera then return end
	setOcclusion(Enum.DevCameraOcclusionMode.Invisicam)
	LocalPlayer.CameraMode = Enum.CameraMode.Classic
	Camera.CameraType = Enum.CameraType.Custom
	setZoom(cameraDistance)
end

local function setThirdPerson(on)
	if on == thirdPersonActive then return end
	thirdPersonActive = on
	if thirdPersonActive then
		Camera = Workspace.CurrentCamera
		local occlusion = Enum.DevCameraOcclusionMode.Zoom
		pcall(function() occlusion = LocalPlayer.DevCameraOcclusionMode end)
		savedSettings = {
			minZoom = LocalPlayer.CameraMinZoomDistance, maxZoom = LocalPlayer.CameraMaxZoomDistance,
			cameraMode = LocalPlayer.CameraMode, occlusion = occlusion,
			cameraType = Camera and Camera.CameraType or Enum.CameraType.Custom,
		}
		pcall(function() RunService:UnbindFromRenderStep(TP_RENDER) end)
		RunService:BindToRenderStep(TP_RENDER, Enum.RenderPriority.Camera.Value - 1, onThirdPerson)
		onThirdPerson()
	else
		pcall(function() RunService:UnbindFromRenderStep(TP_RENDER) end)
		restoreSettings()
	end
end

local function getParts()
	local character = LocalPlayer.Character
	if not character then return nil end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")
	if not humanoid or not root or humanoid.Health <= 0 then return nil end
	return humanoid, root
end

local function onSpin(dt)
	if not spinning then return end
	local humanoid, root = getParts(); if not humanoid then return end
	humanoid.AutoRotate = false
	yaw += spinSpeed * dt
	root.CFrame = CFrame.new(root.Position) * CFrame.Angles(0, yaw, 0)
end

local function setSpin(on)
	if on == spinning then return end
	spinning = on
	local humanoid, root = getParts()
	if spinning and root then local _, cy = root.CFrame:ToOrientation(); yaw = cy end
	if humanoid then humanoid.AutoRotate = not spinning end
end

RunService:BindToRenderStep(SPIN_RENDER, Enum.RenderPriority.Character.Value + 1, onSpin)

local function onFly(dt)
	if not flying then return end
	local character = LocalPlayer.Character; if not character then return end
	local hrp = character:FindFirstChild("HumanoidRootPart"); if not hrp then return end
	local moveDir = Vector3.new(0, 0, 0)
	if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir += Camera.CFrame.LookVector end
	if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir -= Camera.CFrame.LookVector end
	if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir -= Camera.CFrame.RightVector end
	if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir += Camera.CFrame.RightVector end
	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir += Vector3.new(0, 1, 0) end
	if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir -= Vector3.new(0, 1, 0) end
	if moveDir.Magnitude > 0 then moveDir = moveDir.Unit end
	hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
	hrp.CFrame = hrp.CFrame + moveDir * flySpeed * dt
end

local function setFly(on)
	flying = on
	if flying then
		pcall(function() RunService:UnbindFromRenderStep(FLY_RENDER) end)
		RunService:BindToRenderStep(FLY_RENDER, Enum.RenderPriority.Character.Value + 2, onFly)
	else
		pcall(function() RunService:UnbindFromRenderStep(FLY_RENDER) end)
	end
end

local rainbowHue = 0
local prepared = {}

local function prepareCharacter(character)
	if prepared[character] then return end
	prepared[character] = true
	if Config.RainbowRemoveClothes then
		for _, child in character:GetChildren() do
			if child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") then child:Destroy() end
		end
	end
	for _, desc in character:GetDescendants() do
		if desc:IsA("BasePart") and desc.Name ~= "HumanoidRootPart" then
			if desc:FindFirstAncestorOfClass("Tool") then continue end
			if desc:IsA("MeshPart") then desc.TextureID = "" end
			local surface = desc:FindFirstChildOfClass("SurfaceAppearance")
			if surface then surface:Destroy() end
		end
	end
end

local function paintCharacter(character, color)
	prepareCharacter(character)
	local bodyColors = character:FindFirstChildOfClass("BodyColors")
	if bodyColors then
		bodyColors.HeadColor3 = color; bodyColors.TorsoColor3 = color
		bodyColors.LeftArmColor3 = color; bodyColors.RightArmColor3 = color
		bodyColors.LeftLegColor3 = color; bodyColors.RightLegColor3 = color
	end
	for _, desc in character:GetDescendants() do
		if not desc:IsA("BasePart") then continue end
		if desc.Name == "HumanoidRootPart" then continue end
		if desc:FindFirstAncestorOfClass("Tool") then continue end
		desc.Color = color
		local mesh = desc:FindFirstChildOfClass("SpecialMesh")
		if mesh then mesh.VertexColor = Vector3.new(color.R, color.G, color.B) end
	end
end

local function paintTools(character, color)
	for _, tool in ipairs(character:GetChildren()) do
		if tool:IsA("Tool") then
			for _, d in ipairs(tool:GetDescendants()) do
				if d:IsA("BasePart") then
					d.Color = color
					local mesh = d:FindFirstChildOfClass("SpecialMesh")
					if mesh then mesh.VertexColor = Vector3.new(color.R, color.G, color.B) end
				end
			end
		end
	end
end

local function GetClosestPlayer()
	local closestPlayer, closestDistance = nil, Config.AimbotFOV
	local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			local character = player.Character
			if character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("Humanoid") then
				if character.Humanoid.Health > 0 then
					if Config.TeamCheck and player.Team == LocalPlayer.Team and player.Team ~= nil then continue end
					local screenPos, onScreen = Camera:WorldToViewportPoint(character.HumanoidRootPart.Position)
					if onScreen then
						local distance = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
						if distance < closestDistance then closestDistance = distance; closestPlayer = player end
					end
				end
			end
		end
	end
	return closestPlayer
end

local function AimbotLoop()
	if not Config.AimbotEnabled then return end
	local aimKeyDown = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
	if aimKeyDown or UserInputService:IsKeyDown(Config.AimbotKey) then
		local target = GetClosestPlayer()
		if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
			local targetPos = target.Character.HumanoidRootPart.Position
			Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, targetPos), Config.AimbotSmoothness)
		end
	end
end

local function setProp(obj, prop, value) if obj and obj[prop] ~= value then obj[prop] = value end end
local function setLine(line, from, to, color, thickness)
	if not line then return end
	setProp(line, "From", from); setProp(line, "To", to); setProp(line, "Color", color); setProp(line, "Thickness", thickness)
end
local function HideAllObjects(objs)
	if not objs then return end
	if objs.healthBar then setProp(objs.healthBar, "Visible", false) end
	if objs.tracer then setProp(objs.tracer, "Visible", false) end
	if objs.skeleton then for _, l in ipairs(objs.skeleton) do setProp(l, "Visible", false) end end
end
local function ClearESPForPlayer(player)
	if ESPObjects[player] then
		if ESPObjects[player].healthBar then ESPObjects[player].healthBar:Remove() end
		if ESPObjects[player].tracer then ESPObjects[player].tracer:Remove() end
		if ESPObjects[player].skeleton then for _, l in ipairs(ESPObjects[player].skeleton) do l:Remove() end end
		ESPObjects[player] = nil
	end
end

local deadCharacters = {}
local watchCharacterDeath
local function CreateESPForPlayer(player)
	ESPObjects[player] = {}
	local healthBar = Drawing.new("Line"); healthBar.Thickness = 3; healthBar.Color = Color3.fromRGB(0, 255, 0); healthBar.Visible = false
	ESPObjects[player].healthBar = healthBar
	local tracer = Drawing.new("Line"); tracer.Thickness = 1; tracer.Color = Config.ESPColor; tracer.Visible = false
	ESPObjects[player].tracer = tracer
	local skeleton = {}
	for i = 1, #BONE_CONNECTIONS do
		local line = Drawing.new("Line"); line.Thickness = Config.ESPThickness; line.Color = Config.ESPColor; line.Visible = false
		table.insert(skeleton, line)
	end
	ESPObjects[player].skeleton = skeleton
	if player.Character then watchCharacterDeath(player, player.Character) end
	player.CharacterAdded:Connect(function(newChar)
		local newHum = newChar:WaitForChild("Humanoid", 5)
		if newHum then deadCharacters[newChar] = nil; watchCharacterDeath(player, newChar) end
	end)
	player.CharacterRemoving:Connect(function() HideAllObjects(ESPObjects[player]) end)
end

local function IsCharacterValid(player)
	if not player or not player.Character then return false end
	local char = player.Character
	if deadCharacters[char] then return false end
	if not char:IsDescendantOf(Workspace) then return false end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return false end
	if hum.Health <= 0 then return false end
	if hum:GetState() == Enum.HumanoidStateType.Dead then return false end
	return true
end

watchCharacterDeath = function(player, char)
	local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 5)
	if hum then
		hum.Died:Connect(function()
			deadCharacters[char] = true
			if ESPObjects[player] then HideAllObjects(ESPObjects[player]) end
		end)
	end
end

local function UpdateESP()
	if not Config.ESPEnabled then
		for _, objs in pairs(ESPObjects) do HideAllObjects(objs) end
		return
	end
	for player, objs in pairs(ESPObjects) do
		if not IsCharacterValid(player) then HideAllObjects(objs); continue end
		local char = player.Character
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local humanoid = char:FindFirstChild("Humanoid")
		local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
		if Config.TeamCheck and player.Team == LocalPlayer.Team and player.Team ~= nil then HideAllObjects(objs); continue end
		if onScreen then
			local headPos = char:FindFirstChild("Head")
			local headY = headPos and Camera:WorldToViewportPoint(headPos.Position).Y or screenPos.Y - 50
			local legY = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0)).Y
			local boxHeight = math.abs(headY - legY)
			local boxX, boxY = screenPos.X, headY
			if Config.ESPShowHealth and humanoid then
				local hp = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
				setLine(objs.healthBar, Vector2.new(boxX - 15, boxY + boxHeight), Vector2.new(boxX - 15, boxY + boxHeight - boxHeight * hp), Color3.fromRGB(255 - 255 * hp, 255 * hp, 0), 3)
				setProp(objs.healthBar, "Visible", true)
			else setProp(objs.healthBar, "Visible", false) end
			if Config.ESPShowLine and objs.tracer then
				setLine(objs.tracer, Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y), Vector2.new(screenPos.X, screenPos.Y), Config.ESPColor, 1)
				setProp(objs.tracer, "Visible", true)
			else setProp(objs.tracer, "Visible", false) end

local function encodeValue(v)
	if type(v) == "table" then
		local parts = {}
		for k, val in pairs(v) do parts[#parts + 1] = "[" .. encodeValue(k) .. "]=" .. encodeValue(val) end
		return "{" .. table.concat(parts, ",") .. "}"
	elseif type(v) == "string" then return string.format("%q", v)
	elseif type(v) == "number" or type(v) == "boolean" then return tostring(v)
	elseif typeof(v) == "Color3" then return "{__c=true,r=" .. v.R .. ",g=" .. v.G .. ",b=" .. v.B .. "}"
	elseif typeof(v) == "EnumItem" then return "{__e=true,t=" .. string.format("%q", tostring(v.EnumType)) .. ",n=" .. string.format("%q", v.Name) .. "}"
	end
	return "nil"
end

local function convertDecoded(t)
	if type(t) ~= "table" then return t end
	if t.__c then return Color3.new(t.r, t.g, t.b) end
	if t.__e then return Enum[t.t] and Enum[t.t][t.n] end
	for k, v in pairs(t) do t[k] = convertDecoded(v) end
	return t
end
